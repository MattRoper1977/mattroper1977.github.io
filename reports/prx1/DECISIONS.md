```python
"""Run real publication generators; compare every actual header/main operation."""

from collections import Counter
import hashlib, importlib, inspect, json, re, subprocess, sys, datetime, traceback

import structural_html as helper
import build_education
original=helper.replace_first_element
records=[]
def digest(text):return hashlib.sha256(text.encode()).hexdigest()
def compare(source,tag,replacement):
    frame=inspect.currentframe().f_back
    path=frame.f_locals.get('path',frame.f_locals.get('page'))
    assert path is not None, frame.f_code.co_filename
    key=f'{len(records):02d}_{tag}'
    row={'path':str(path.relative_to(OUT)),'tag':tag,'input_sha256':digest(source),
         'replacement_sha256':digest(replacement),'raw_header_closes':source.count('</header>'),
         'raw_main_closes':source.count('</main>')}
    (EVIDENCE/(key+'_input.html')).write_text(source)
    (EVIDENCE/(key+'_replacement.html')).write_text(replacement)

    (EVIDENCE/(key+'_old.html')).write_text(old)
    try:new,newcount=original(source,tag,replacement)
    except Exception as e:
        row['raise']=repr(e);records.append(row)
        (EVIDENCE/'records.json').write_text(json.dumps(records,indent=2)+'\n')
        raise
    (EVIDENCE/(key+'_new.html')).write_text(new)
    row.update(old_count=count,new_count=newcount,old_sha256=digest(old),new_sha256=digest(new),equal=old==new)
    records.append(row)
    (EVIDENCE/'records.json').write_text(json.dumps(records,indent=2)+'\n')
    assert count==newcount==1 and old==new, row
    return new,newcount
```

```python
real=structural_html.replace_first_element
records=[]
def old(source,tag,replacement):
    return re.subn(fr'<{tag}\b[^>]*>.*?</{tag}>',lambda _:replacement,source,count=1,flags=re.S)
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def tree(root):return {str(p.relative_to(root)):sha(p) for p in root.rglob('*') if p.is_file()}
def fixture(source,dest):
    dest.mkdir(parents=True,exist_ok=True)
    for group in ['education-site','education-lessons','education-apps']:
        (dest/group/'assets').mkdir(parents=True,exist_ok=True)
        for p in (source/group).rglob('*.html'):
            q=dest/p.relative_to(source);q.parent.mkdir(parents=True,exist_ok=True);shutil.copyfile(p,q)

def wrap(name,func):
    def run(output,*args):
        pre=EV/(name+'-pre');fixture(output,pre)
        oldout=EV/(name+'-old');fixture(pre,oldout)
        newresult=func(output,*args)
        structural_html.replace_first_element=old
        try:oldresult=func(oldout,*args)
        finally:structural_html.replace_first_element=real
        expected=tree(oldout)
        differences=[p for p,d in expected.items() if not (output/p).is_file() or sha(output/p)!=d]
        assert not differences,differences
        assert oldresult==newresult,(oldresult,newresult)
        tag='main' if name=='audience' else 'header'
        data=json.loads((EV/'records.json').read_text())
        first=next(x for x in data if x['tag']==tag)['path']
        source=(pre/first).read_text()
        opening=re.search(fr'<{tag}\b[^>]*>',source);assert opening
        plant=source[:opening.end()]+f'<{tag} id="prx1-plant">inner</{tag}>'+source[opening.end():]
        assert plant.count('id="prx1-plant"')==1
        assert plant.count(f'</{tag}>')==source.count(f'</{tag}>')+1
        plants={}
        for mode,helper in [('old',old),('repaired',real)]:
            dst=EV/(name+'-'+mode+'-plant');fixture(pre,dst)
            (dst/first).write_text(plant)
            structural_html.replace_first_element=helper
            error=None
            try:func(dst,*args)
            except Exception as e:error=str(e)
            finally:structural_html.replace_first_element=real
            saved=(dst/first).read_text()
            (EV/(name+'-'+mode+'-plant-output.html')).write_text(saved)
            if mode=='repaired':
                assert error and f'Nested <{tag}>' in error,error
                assert saved==plant,'repaired caller wrote the unsafe page'
            else:
                assert error is None,error
                assert saved!=plant and 'id="prx1-plant"' not in saved
                # Lazy replacement leaves the original outer close as an orphan.
                assert saved.count(f'</{tag}>')>saved.count(f'<{tag} ')+saved.count(f'<{tag}>'),first
            plants[mode]={'error':error,'input_sha256':hashlib.sha256(plant.encode()).hexdigest(),
                          'output_sha256':hashlib.sha256(saved.encode()).hexdigest(),
                          'unsafe_page_unchanged':saved==plant}
        records.append({'function':name,'compared_files':len(expected),'differences':differences,'result_equal':True,'planted_path':first,'plant_self_verified':True,'plants':plants})
        (EV/'actual_functions.json').write_text(json.dumps(records,indent=2)+'\n')
        return newresult
    return run
build_education.refresh_audiences=wrap('audience',build_education.refresh_audiences)
build_education.refresh_navigation=wrap('navigation',build_education.refresh_navigation)
result=build_education.build(OUT,ROOT/'lessons',ROOT/'apps',False)
print(json.dumps({'measured_at':datetime.datetime.now(datetime.timezone.utc).isoformat(),'records':records,'build_status':result['status']},indent=2))
```
